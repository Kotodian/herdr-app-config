+// NetworkObserver may be implemented by a host Dialer to supply a stable local
+// network revision. It must honor ctx and must not probe business destinations.
+type NetworkObserver interface {
+	NetworkRevision(context.Context) (string, error)
+}
+
+func (m *session) networkRevision(ctx context.Context) (string, error) {
+	if source, ok := m.dependencies.Dialer.(NetworkObserver); ok {
+		return source.NetworkRevision(ctx)
+	}
+	return localNetworkRevision(ctx)
+}
+
+func (m *session) watchNetwork() {
+	ticker := time.NewTicker(time.Second)
+	defer ticker.Stop()
+	previous := m.networkRevisionValue
+	for {
+		select {
+		case <-m.ctx.Done():
+			return
+		case <-ticker.C:
+		}
+		ctx, cancel := context.WithTimeout(m.ctx, 4*time.Second)
+		revision, err := m.networkRevision(ctx)
+		cancel()
+		if err != nil || revision == previous {
+			continue
+		}
+		previous = revision
+		select {
+		case <-m.changes:
+		default:
+		}
+		select {
+		case m.changes <- revision:
+		case <-m.ctx.Done():
+			return
+		}
+	}
+}
+
+func (m *session) resetNetwork() error {
+	m.healthMu.Lock()
+	m.networkCancel()
+	m.healthMu.Unlock()
+	next := &session{config: m.config, dependencies: m.dependencies}
+	if err := next.prepareRoutes(); err != nil {
+		return errors.Join(err, next.closeRoutes())
+	}
+	m.healthMu.Lock()
+	m.networkCtx, m.networkCancel = context.WithCancel(m.ctx)
+	for index, replacement := range next.routes {
+		replacement.ctx = m.networkCtx
+		replacement.blocked = m.routes[index].blocked
+		for host, until := range m.routes[index].cooldowns {
+			if time.Now().Before(until) {
+				replacement.cooldowns[host] = until
+			}
+		}
+	}
+	m.retired = append(m.retired, m.routes...)
+	m.routes, m.generation = next.routes, next.generation
+	m.healthMu.Unlock()
+	m.pool.advance(next.generation)
+	m.reapRoutes(false)
+	return nil
+}
+
+func (m *session) commitRequest(ctx context.Context, candidate *route, stopNetwork func() bool) error {
+	m.healthMu.Lock()
+	defer m.healthMu.Unlock()
+	if candidate.generation != m.generation || (candidate.ctx != nil && candidate.ctx.Err() != nil) {
+		return context.Canceled
+	}
+	if err := ctx.Err(); err != nil {
+		return err
+	}
+	if !stopNetwork() {
+		return context.Canceled
+	}
+	return nil
+}
+

+		}
+	}
