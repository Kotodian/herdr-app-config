+type ProbeWindow struct {
+	Completed   uint64
+	Success     uint64
+	Canceled    uint64
+	SuccessRate *float64
+	Truncated   bool
+}
+
+type ProbeEndpointDiagnostics struct {
+	Routes []int
+	Window ProbeWindow
+}

+type BusinessObservation struct {
+	Route             int
+	Generation        uint64
+	Protocol          string
+	Stage             string
+	Reused            bool
+	FirstByte         *time.Duration
+	FirstByteBoundary string
+	Elapsed           time.Duration
+	BodyBytes         uint64
+	Interrupted       bool
+	Canceled          bool
+	TimedOut          bool
+	Status            int
+	Completed         time.Time
+}
+
+type Diagnostics struct {
+	Generation         uint64
+	Window             time.Duration
+	CapacityPerOutcome int
+	Probes             []ProbeEndpointDiagnostics
+	Business           []BusinessObservation
+	CacheAvailable     bool
+}
+
+func (h *accessHealth) diagnostics(now time.Time) ProbeWindow {
+	var window ProbeWindow
+	for _, sample := range h.samples {
+		if now.Sub(sample.completed) < 60*time.Second {
+			window.Completed++
+			if sample.success {
+				window.Success++
+			}
+		}
+	}
+	for _, completed := range h.canceledSamples {
+		if now.Sub(completed) < 60*time.Second {
+			window.Canceled++
+		}
+	}
+	if window.Completed > 0 {
+		ratio := float64(window.Success) / float64(window.Completed)
+		window.SuccessRate = &ratio
+	}
+	window.Truncated = !h.evicted.IsZero() && now.Sub(h.evicted) < 60*time.Second
+	return window
+}
+
+func (m *Mode) Diagnostics() Diagnostics { return m.current().diagnostics(time.Now()) }
+
+func (m *session) diagnostics(now time.Time) Diagnostics {
+	m.healthMu.Lock()
+	defer m.healthMu.Unlock()
+	result := Diagnostics{Generation: m.generation, Window: 60 * time.Second, CapacityPerOutcome: 256}
+	positions := make(map[*accessHealth]int)
+	for _, candidate := range m.routes {
+		if candidate.access == nil {
+			continue
+		}
+		index, exists := positions[candidate.access]
+		if !exists {
+			index = len(result.Probes)
+			positions[candidate.access] = index
+			result.Probes = append(result.Probes, ProbeEndpointDiagnostics{Window: candidate.access.diagnostics(now)})
+		}
+		result.Probes[index].Routes = append(result.Probes[index].Routes, candidate.id)
+	}
+	for _, event := range m.business {
+		if event.Generation == m.generation && now.Sub(event.Completed) < 60*time.Second {
+			copy := event
+			if event.FirstByte != nil {
+				elapsed := *event.FirstByte
+				copy.FirstByte = &elapsed
+			}
+			result.Business = append(result.Business, copy)
+		}
+	}
+	return result
+}
+
+func (m *session) recordBusiness(event BusinessObservation) {
+	m.healthMu.Lock()
+	defer m.healthMu.Unlock()
+	if event.Generation != m.generation {
+		return
+	}
+	if len(m.business) == 256 {
+		copy(m.business, m.business[1:])
+		m.business = m.business[:255]
+	}
+	m.business = append(m.business, event)
+}

+import (
+	"cmp"
+	"net/netip"
+	"slices"
+	"time"

+type DirectAddress struct {
+	IP       netip.Addr
+	Failures int
+	Elapsed  time.Duration
+}
+
+type DirectAddressStrategy interface {
+	Order(target string, generation uint64, addresses []DirectAddress)
+	Feedback(target string, generation uint64, ip netip.Addr, elapsed time.Duration, err error)
+}
+
+type DirectAddressProvider interface {
+	DirectAddresses() DirectAddressStrategy
+}
+
+type HistoricalAddresses struct{}
+
+func (HistoricalAddresses) Order(_ string, _ uint64, addresses []DirectAddress) {
+	slices.SortStableFunc(addresses, func(left, right DirectAddress) int {
+		if order := cmp.Compare(left.Failures, right.Failures); order != 0 {
+			return order
+		}
+		if left.Elapsed == 0 && right.Elapsed != 0 {
+			return 1
+		}
+		if left.Elapsed != 0 && right.Elapsed == 0 {
+			return -1
+		}
+		return cmp.Compare(left.Elapsed, right.Elapsed)
+	})
+}

+func (HistoricalAddresses) Feedback(string, uint64, netip.Addr, time.Duration, error) {}

+# 地址策略与本地诊断
+
+本地诊断不创建监控服务、后台原站测速或额外 HTTP 请求。
+开发阶段只编译测试源码；本文示例与测试没有在 dev 执行。
+
+## 地址策略
+
+Rust `DirectOutbound<S>` 的 S 是 `DirectAddressStrategy`，底层 `TcpDialer` 独立保存：
+
+1. `resolve(dialer, domain, port)` 为真实业务提供地址，默认调用宿主 dialer 的解析能力。
+2. `order(domain, generation, &mut [DirectAddress])` 接收本代的地址历史快照。
+3. 每次已完成且未取消的 TCP 建连调用 `feedback`，带 domain、generation、IP:port 和真实耗时/错误。
+
+NativeFactory 通过既有 dialer 依赖的默认方法 `TcpDialer::direct_addresses()` 取得策略。
+宿主可覆盖此方法返回自定义策略，再沿原 `Dependencies::with_dialer` 注入；
+原 dialer 实现无需增加方法，默认仍为 `HistoricalAddresses`。
+TCP 经 NativeStream，HTTP 经 NativeHttp connector，均实际调用同一 DirectOutbound。
+HTTP 复用连接时不重新解析，新连接由策略提供地址；反代的原 DNS 缓存路径保持独立。
+TLS SNI、authority 和目标头仍由原域名映射确定。
+
+Go 沿用 `Dependencies.Resolver` 提供地址来源。宿主 Dialer 可额外实现小接口
+`DirectAddressProvider.DirectAddresses() DirectAddressStrategy`，替换 Order/Feedback；
+未实现或返回 nil 则为 HistoricalAddresses。Factory → prepareRoutes → connectRoute →
+dialRoute 同时服务 Connect 与 HTTP 新连接，没有强行复制 Rust 泛型。
+
+内建策略保留解析顺序作为初始顺序，稳定排序时先比较失败次数，
+再优先成功历史中耗时较好的地址，不建立新的评分算法。
+DNS、全部地址建连、TLS 和跨出口尝试仍共享既有执行器 deadline。
+仅 TCP 建连未发送业务数据时轮换地址，不因目标响应或正文失败重放。
+网络变化创建新代直连历史；旧对象迟到反馈不会写入新对象。
+Go 反馈也检查获取上下文、route 和 session generation。
+
+策略同步方法不在健康锁内调用，须快速、可并发调用；异步来源必须遵守取消。
+外部策略若自行保存历史，必须按传入 generation 隔离，不能把旧代反馈当成新网络测量。
+
+## 探测窗口
+
+Rust 通过 Observer 的默认回调 `probe_diagnostics` 交付只读
+`Arc<dyn ProbeDiagnosticsSource>`，自定义 observer 可保留它并调用 snapshot。
+`DiagnosticRecorder::snapshot().probes` 和 `snapshot_if_changed` 已消费此入口。
+源通过弱引用读取当前 Registry，不延长实例资源寿命。
+读取时过滤过期样本，窗口过期或 generation 改变会更新 recorder revision。
+Go 使用 `Mode.Diagnostics()` 读取生产 probeBatch/probeLoop 写入的 accessHealth 副本，
+原 Status 字段与行为不变。
+
+| 字段 | 两种实现共用的含义 |
+| --- | --- |
+| generation / Generation | 当前网络代际，网络变化后重新开始 |
+| endpoints / Probes | 仅实际接入点；同地址共用探测的线路编号合并到 routes/Routes，不重复累计 |
+| completed / Completed | 读取时最近 60 秒内完成的成功、失败和超时 |
+| success / Success | 上述完成样本的成功数 |
+| canceled / Canceled | 独立的最近 60 秒取消数，不进入完成分母 |
+| success_rate / SuccessRate | success/completed；零分母为 None/nil，不是 0% |
+| truncated / Truncated | 窗口内样本因容量被逐出，计数仅代表保留样本 |
+| capacity_per_outcome / CapacityPerOutcome | 完成和取消各最多 256 条；成功为完成样本的子集 |
+
+C-only 的接入点列表为空，不生成原站测量。
+Rust 没有注册源、实例已释放或当代已失效时 probes 为 None。
+读取不改变资格、选择或熔断；真实业务接入结果不进入探测窗口。
+诊断以线路编号关联宿主持有的配置，不输出带凭据节点 URL。

+## 业务指标与真实口径
+
+| 指标 | Rust 生产写入 → 消费 | Go 生产写入 → 消费 |
+| --- | --- | --- |
+| 首字节 | managed_stream::ObservedStream 首次写入后开始、首次读到字节结束的 UpstreamFirstByte → recorder events/metrics；可能是 TLS 记录 | Request 提交后开始；HTTP/1 在 reader 首次可读字节处采集，HTTP/2 用固定依赖的 GotFirstResponseByte 回调；FirstByteBoundary 标明 response-byte 或 http2-response-headers |
+| HTTP 响应头 | upstream_request_filter → upstream_response_filter 的 HttpHeaders；不是精确 HTTP 首字节 | Business 的 headers.Elapsed；交换失败也保留 interrupted 记录 |
+| 正文数据/中断 | upstream_response_filter 创建 HttpBody、response_body_filter 累计数据、logging/Operation Drop 结算 → events、metrics、resources.incomplete_body_bytes | responseBody.Read 计真实字节、EOF、错误，Close 结算 body 事件；提前关闭标 Interrupted |
+| 缓存 | request_cache_filter 的实际 cache-hit 回调 → HttpCacheHit 字节/耗时，与上游 HttpBody 分开 | 没有响应缓存，CacheAvailable=false；不表示命中率 0% |
+| 复用 | connected_to_upstream 的 reused → HttpReuse/Http2Reuse → recorder | pool.take 实际命中 → connectHTTP → Business.Reused；单次请求标记，不写入共享连接 |
+
+Rust 精确逐请求 HTTP 首字节尚无可用采集边界，现有 Pingora 回调给的是完整响应头。
+Go 固定版本的 HTTP/2 trace 在完整首个 HEADERS 处理时调用，不能称作线上第一个字节；
+因此明确标为 http2-response-headers。Go 原始 TCP 首字节/字节吞吐诊断尚未实现。
+这些限制不通过用响应头或 TLS 握手耗时冒充缺失测量来销项。
+
+Rust 业务 counters/resources 为会话累积，events 为最近 256 条事件，
+与新增的 60 秒探测窗口独立。默认不输出目标名，目标捕获沿原 opt-in。
+Go Business 为最近 60 秒内保留的最多 256 条 headers/body 记录，带出口编号、协议、代际，
+不包含 URL/正文。正文在调用方关闭 Body 时结算，活跃正文尚无最终观测。
+TimedOut 保留 SDK 预算、调用方 deadline 和实际 I/O 超时；Canceled 仅表示调用方/实例取消，
+不会因 SDK 预算内部使用 cancelRequest 终止 HTTP/2 而冒充取消。
+HTTP/1 上传失败触发的立即读 deadline 只是唤醒读者，诊断通过同步保存的写失败原因区分，
+不会把普通上传错误变成超时。
+旧代正文仍能读取，但迟到记录不进入当前代诊断；Restart 的新句柄读取新 session。
+
+正文 bytes 与 elapsed 是被动业务数据，不是主动带宽测试或可跨目标排名的吞吐值；
+编码、缓存和客户端背压都会影响它们。dial_elapsed 是拨号总成本，不是 RTT；
+探测失败比例不是 IP 丢包率。没有合成未测量的 RTT、丢包率或吞吐指标。
+

+### 地址组合
+
+原文 §2：“DirectOutbound<S> … 候选地址来源与地址选择策略”；§10：“从真实业务需要的候选地址中选择”；
+§11：“解析或读取候选 → 优先使用历史表现好的地址 → 在允许的建连失败边界尝试其他地址 → 记录本次结果”。
+
+起点事实：Rust S 是 TcpDialer，connect_addresses 内部固定排序；Go 有 Resolver，排序仍在 dialRoute 内。
+本次新增 Rust DirectAddressStrategy（resolve/order/feedback）与 HistoricalAddresses，
+S 是策略，dialer 独立。通过带默认实现的 TcpDialer::direct_addresses 接入 Native 组装，
+不增加 Dependencies 必填字段。Go 保留 Resolver，通过可选 DirectAddressProvider 取得策略，
+与 Dialer 原接口兼容。
+
+生产链：Rust build_routes → Backend::create_in_generation → NativeStream/NativeHttp →
+DirectOutbound → 地址策略 → 实际拨号 → 本代历史/反馈。HTTP 新建连接也使用策略来源；
+池命中不额外解析。Go Create/prepareRoutes → Connect 或 Request → connectRoute/dialRoute →
+Resolver/Order → 真实拨号 → 当前代反馈。策略不在健康锁内执行。
+
+默认按失败数和真实成功耗时稳定排序，不后台测原站、不追加 TLS/HTTP 验证。
+地址轮换仍在已有执行器同一 deadline 与取消边界内，既有 SNI/authority 和提交后不重放保持。
+Rust 每代新对象，Go 每代 route.direct；自定义共享策略收到 generation，必须自行隔离其私有历史。

+### 探测窗口和宿主消费
+
+原文 §6.3：“最近 60 秒成功次数 / 完成次数，失败与超时进入分母，取消单列”；
+“诊断与后续阈值验证；少量启动样本不做百分比熔断”。
+
+起点有 60 秒、最多 256 条完成样本和单列累计取消，但 Rust Observer/Go Status 都不能消费窗口。
+本次保留生产 Health/accessHealth 写入链，增加各最多 256 条的近期取消队列；
+完成/成功和取消按读取时刻过滤，容量逐出显式 truncated，零分母 None/nil。
+Rust Observer 兼容默认回调交付 Weak Registry-backed source，DiagnosticRecorder.snapshot.probes 读取当前代；
+Go Mode.Diagnostics 直接复制当前代 accessHealth。重复接入地址按线路编号分组合并，不重计。
+C-only 返回空接入点列表，没有原站样本。读窗口不会影响熔断或同层选择。

+真实写入仍是 Rust check_routes_until → ProbeResult.completed → Health.record_at，
+Go probeBatch/probeLoop → recordProbe。业务 record_outcome/record 不写探测样本，
+取消不进入 completed，旧代反馈不污染新代。网络失效时 Rust source 返回 None，
+发布后读取新 Registry；Go 锁内读取 current session/routes。
+同端点的后台探测即使只由其中一条线路到期触发，也向全部关联 Health 记录同一完成结果，
+因此不同 jitter/due 不会使导出的首条线路窗口漏掉其他线路触发的真实探测。

+### 业务诊断
+
+原文 §6.3：“首字节、正文吞吐与中断 … 真实业务，缓存命中与连接复用单列”；
+“首版不拿不同目标、响应大小的样本直接排名”。
+
+Rust 现有生产链已完整的部分不重做：ObservedStream → UpstreamFirstByte，
+Pingora HttpHeaders、HttpBody、HttpCacheHit、HttpReuse/Http2Reuse → Observer →
+DiagnosticRecorder events/metrics/resources。缓存路径不混作上游正文，
+被动字节流首字节可能是 TLS 记录，不是 HTTP TTFB。
+已有资源/缓存/复用测试保留；memory/disk cache 条件测试不在默认 feature 编译覆盖之内。
+
+Go 起点没有对应业务诊断消费链。本次 Request 在真实提交后开始计时，
+HTTP/1 reader.Peek(1)、HTTP/2 固定依赖 trace 记录响应边界，
+pool.take 实际命中标记单次复用，responseBody.Read/Close 记录真实正文 bytes/elapsed、
+EOF、错误、提前关闭。Mode.Diagnostics.Business 返回最近 60 秒内最多 256 条记录，
+带 route/generation/protocol，不含 URL/正文；旧代迟到事件不进入当前代。
+没有缓存实现，CacheAvailable=false，不能声称命中率为零。
+Go 的 TimedOut 与 Canceled 分别保留超时和调用方/实例取消，
+SDK 预算触发内部 cancelRequest 不被写成业务取消；
+`TestHTTPDiagnosticsSeparateSDKTimeoutFromCallerCancellation` 包含 HTTP/1、HTTP/2 两种来源对比，
+只编译，不执行。Rust 窗口集成用例也覆盖共享接入点不同 due 下的失败与取消。
+同轮进一步修正 HTTP/1 上传失败通过立即读 deadline 唤醒读者的诊断口径：
+原写失败是否超时通过 atomic 状态同步传递，避免把普通上传错误归为超时；
+`TestHTTP1UploadFailureWakeupIsNotRecordedAsTimeout` 为对应测试源码，未执行。
+
+准确限制：Rust 精确逐 HTTP 请求首字节尚未支持，只有完整响应头耗时和字节流首字节；
+Go HTTP/2 trace 的真实边界是首个完整 HEADERS，显式 FirstByteBoundary=http2-response-headers，
+不是线上首字节；Go 原始 TCP 首字节/吞吐诊断尚未支持。
+正文数据是被动原始观测，未新增自动吞吐值、RTT、IP 丢包率或跨目标评分。
+无需真实节点的读法和具体口径见 diagnostics.md。
