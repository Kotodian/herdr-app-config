+`payload: Http` 表示 SDK 接收可解析的 HTTP，并为接管的 TLS 连接终止本地 TLS，
+因此 `configuration_policy()` 要求本地 CA。`payload: Tcp` 保留客户端原始字节，
+不要求本地 CA；普通 HTTP 反代会在该入口下被过滤。EdgeOne WebSocket 仅支持 TCP。
+HTTP 经隧道出口时，SDK 在隧道内为映射后的真实目标建立独立的上游 TLS/HTTP 连接。
+直连同样使用白名单的目标域名映射，目的 IP 不作为 TLS 身份。
+
+候选固定按 Relay → CDN → 直连 → 隧道尝试。同层先比较连续接入失败数，再比较
+`dial_elapsed` 的 EWMA（新样本占 20%），最快 20% 范围按目标与会话稳定分配。
+已知下载域名、`/depot/`、`/client/` 和下载文件后缀在发送前分类；
+Cloudflare Worker 与普通 EdgeOne HTTP CDN 不承载这类已识别下载。
+HTTP 请求可能开始发送后不跨候选重放，包括 GET、429、403、404 和服务器错误。
+同一个下游 H2 连接内的请求分别选路；上游池按出口、实例 generation、TLS 身份、

+当前 Pingora 版本的池复用要求真实 socket 描述符，因此 HTTP 直连和 HTTP 隧道在

+        self.connect_addresses(domain, addresses, observation).await
+    }
+
+    pub async fn connect_addresses(
+        &self,
+        domain: &str,
+        mut addresses: Vec<SocketAddr>,
+        observation: &ObservationContext,
+    ) -> Result<BoxStream> {
+        {
+            let health = self.health.lock();
+            addresses.sort_by_key(|address| {
+                let state = health.get(&(domain.into(), *address, self.generation));
+                (
+                    state.map_or(0, |state| state.failures),
+                    state
+                        .and_then(|state| state.elapsed)
+                        .unwrap_or(Duration::MAX),
+                )
+            });
+        }
+        let mut last = Error::NoEndpoint;
+        for address in addresses {
+            let started = Instant::now();
+            let result = self
+                .source
+                .connect_observed(&address.ip().to_string(), address.port(), observation)
+                .await;
+            if matches!(result, Err(Error::Canceled)) {
+                return result;
+            }
+            {
+                let mut health = self.health.lock();
+                let state = health
+                    .entry((domain.into(), address, self.generation))
+                    .or_default();
+                match &result {
+                    Ok(_) => {
+                        state.failures = 0;
+                        state.elapsed = Some(started.elapsed());
+                    }
+                    Err(_) => state.failures += 1,
+                }
+            }
+            match result {
+                Ok(stream) => return Ok(stream),
+                Err(error) => last = error,
+            }
+        }
+        Err(last)
+    }
+}

+#[cfg(feature = "reverse-proxy")]
+impl SocketBridge {
+    pub async fn new() -> Result<Self> {
+        let listener = tokio::net::TcpListener::bind((std::net::Ipv4Addr::LOCALHOST, 0))
+            .await
+            .map_err(|_| Error::Transport)?;
+        Ok(Self {
+            address: listener.local_addr().map_err(|_| Error::Transport)?,
+            listener: tokio::sync::Mutex::new(Some(listener)),
+            tasks: tokio_util::task::TaskTracker::new(),
+            stop: tokio_util::sync::CancellationToken::new(),
+        })
+    }
+
+    pub async fn connect(
+        &self,
+        mut upstream: BoxStream,
+        sockets: Arc<crate::stream::Connections>,
+        cancel: tokio_util::sync::CancellationToken,
+    ) -> Result<pingora_core::protocols::l4::stream::Stream> {
+        let listener = self.listener.lock().await;
+        let listener = listener.as_ref().ok_or(Error::Canceled)?;
+        if self.stop.is_cancelled() || cancel.is_cancelled() {
+            return Err(Error::Canceled);
+        }
+        let client = tokio::net::TcpStream::connect(self.address)
+            .await
+            .map_err(|_| Error::Transport)?;
+        client.set_nodelay(true).map_err(|_| Error::Transport)?;
+        let expected = client.local_addr().map_err(|_| Error::Transport)?;
+        let server = loop {
+            let (stream, address) = tokio::select! {
+                biased;
+                _ = self.stop.cancelled() => return Err(Error::Canceled),
+                result = listener.accept() => result.map_err(|_| Error::Transport)?,
+            };
+            if address == expected {
+                break stream;
+            }
+        };
+        server.set_nodelay(true).map_err(|_| Error::Transport)?;
+        let mut server = sockets.track(Box::new(server))?;
+        let mut client = sockets.track_native(client)?;
+        let lifetime = tokio_util::sync::CancellationToken::new();
+        client.tracer = Some(pingora_core::upstreams::peer::Tracer(Box::new(
+            BridgeTrace {
+                native: client.tracer.take(),
+                cancel: lifetime.clone(),
+            },
+        )));
+        let stop = self.stop.clone();
+        self.tasks.spawn(async move {
+            tokio::select! {
+                biased;
+                _ = stop.cancelled() => {}
+                _ = cancel.cancelled() => {}
+                _ = lifetime.cancelled() => {}
+                _ = sockets.cancelled() => {}
+                _ = tokio::io::copy_bidirectional(&mut server, &mut upstream) => {}
+            }
+        });
+        Ok(client)
+    }
+
+    pub async fn close(&self) {
+        self.stop.cancel();
+        self.listener.lock().await.take();
+        self.tasks.close();
+        self.tasks.wait().await;
+    }
+}

+impl<P: Probe + 'static> ProbeScheduler<P>
+where
+    P::Target: Clone + 'static,
+{
+    pub async fn run(
+        &self,
+        targets: Vec<P::Target>,
+        deadline: Instant,
+        cancel: &CancellationToken,
+    ) -> Result<Vec<ProbeResult<P::Target>>> {
+        let semaphore = Arc::new(tokio::sync::Semaphore::new(self.concurrency));
+        let mut pending = tokio::task::JoinSet::new();
+        for target in targets {
+            let runner = self.runner.clone();
+            let semaphore = semaphore.clone();
+            let cancel = cancel.clone();
+            pending.spawn(async move {
+                let operation = async {
+                    let _permit = tokio::time::timeout_at(deadline, semaphore.acquire_owned())
+                        .await
+                        .map_err(|_| Error::Transport)?
+                        .map_err(|_| Error::Canceled)?;
+                    runner.run(&target, deadline, &cancel).await
+                };
+                let result = tokio::select! {
+                    biased;
+                    _ = cancel.cancelled() => Err(Error::Canceled),
+                    result = operation => result,
+                };
+                ProbeResult {
+                    target,
+                    result,
+                    completed: Instant::now(),
+                }
+            });
+        }
+        let mut results = Vec::new();
+        while let Some(result) = pending.join_next().await {
+            results.push(result.map_err(|_| Error::Task)?);
+        }
+        Ok(results)
+    }

+pub(crate) struct Health {
+    pub eligibility: Eligibility,
+    pub failures: u32,
+    pub successes: u32,
+    pub dial_elapsed: Option<Duration>,
+    pub evidence: Instant,
+    updated: Instant,
+    pub due: Instant,
+    pub backoff: u32,
+    pub isolated: bool,
+    pub targets: HashMap<String, Instant>,
+    pub generation: u64,
+    pub samples: VecDeque<(Instant, bool)>,
+    pub cancellations: u64,

+impl Health {
+    fn usable(&self, route: &Route, traffic: &Traffic<'_>, now: Instant) -> bool {
+        route.payload == traffic.payload
+            && (!traffic.download || route.downloads)
+            && self.eligibility == Eligibility::Available
+            && !self.isolated
+            && (route.endpoint.is_none()
+                || now.duration_since(self.evidence) < Duration::from_secs(60))
+            && !self
+                .targets
+                .get(traffic.host)
+                .is_some_and(|until| *until > now)
+    }
+
+    fn new(generation: u64) -> Self {
+        Self {
+            eligibility: Eligibility::Unavailable,
+            failures: 0,
+            successes: 0,
+            dial_elapsed: None,
+            evidence: Instant::now(),
+            updated: Instant::now(),
+            due: Instant::now(),
+            backoff: 0,
+            isolated: false,
+            targets: HashMap::new(),
+            generation,
+            samples: VecDeque::new(),
+            cancellations: 0,
+        }
+    }
+
+    fn record(&mut self, result: &Result<Duration>, startup: bool, generation: u64) {
+        self.record_at(result, startup, generation, Instant::now());
+    }
+
+    fn record_at(
+        &mut self,
+        result: &Result<Duration>,
+        startup: bool,
+        generation: u64,
+        now: Instant,
+    ) {
+        if self.generation != generation || now < self.updated {
+            return;
+        }
+        if matches!(result, Err(Error::Canceled)) {
+            self.cancellations += 1;
+            return;
+        }
+        self.updated = now;
+        while self
+            .samples
+            .front()
+            .is_some_and(|(time, _)| now.duration_since(*time) >= Duration::from_secs(60))
+        {
+            self.samples.pop_front();
+        }
+        self.samples.push_back((now, result.is_ok()));
+        match result {
+            Ok(elapsed) => {
+                self.failures = 0;
+                self.successes += 1;
+                self.evidence = now;
+                self.dial_elapsed = Some(
+                    self.dial_elapsed
+                        .map_or(*elapsed, |old| old.mul_f64(0.8) + elapsed.mul_f64(0.2)),
+                );
+                if startup || self.successes >= 2 {
+                    self.eligibility = Eligibility::Available;
+                    self.backoff = 0;
+                }
+                self.due = now + jitter();
+            }
+            Err(_) => {
+                self.successes = 0;
+                self.failures += 1;
+                if startup || self.failures >= 3 {
+                    self.eligibility = Eligibility::Unavailable;
+                }
+                let seconds = if self.eligibility == Eligibility::Unavailable {
+                    let seconds = (15u64 << self.backoff.min(3)).min(120);
+                    self.backoff = (self.backoff + 1).min(3);
+                    seconds
+                } else {
+                    15
+                };
+                self.due = now + Duration::from_secs(seconds);
+            }
+        }
+    }
+}

+impl<E: Executor, O: OutcomeClassifier> AttemptLoop<E, O> {
+    pub async fn run(
+        &self,
+        registry: &Registry,
+        traffic: &Traffic<'_>,
+        attempt: &mut Attempt,
+        cancel: &CancellationToken,
+    ) -> Result<E::Output> {
+        loop {
+            let (index, backend) = registry.select(traffic, attempt)?;
+            if !registry.usable(index, traffic) {
+                continue;
+            }
+            let result = tokio::select! {
+                biased;
+                _ = cancel.cancelled() => Err(Error::Canceled),
+                result = tokio::time::timeout_at(attempt.deadline, self.executor.execute(&backend)) =>
+                    result.unwrap_or_else(|_| Err(Error::Operation(crate::Failure::new(
+                        crate::Stage::Connect, crate::FailureCode::Timeout)))),
+            };
+            match result {
+                Ok(output) => {
+                    attempt.committed = true;
+                    return Ok(output);
+                }
+                Err(error) => {
+                    registry.failure(index, traffic.host, &error);
+                    if !self.outcomes.retry(&error, attempt.committed) {
+                        return Err(error);
+                    }
+                }
+            }
+        }
+    }
+}

+impl Planner {
+    pub fn select(
+        &self,
+        traffic: &Traffic<'_>,
+        routes: &[Route],
+        health: &[Health],
+        attempt: &Attempt,
+        session: u64,
+    ) -> Result<usize> {
+        let now = Instant::now();
+        if attempt.committed || now >= attempt.deadline {
+            return Err(Error::NoEndpoint);
+        }
+        for tier in [Tier::A, Tier::B, Tier::C, Tier::D] {
+            let candidates: Vec<_> = routes
+                .iter()
+                .enumerate()
+                .filter(|(index, route)| {
+                    let state = &health[*index];
+                    route.tier == tier
+                        && !attempt.tried.contains(index)
+                        && state.usable(route, traffic, now)
+                })
+                .collect();
+            let Some(failures) = candidates
+                .iter()
+                .map(|(index, _)| health[*index].failures)
+                .min()
+            else {
+                continue;
+            };
+            let candidates: Vec<_> = candidates
+                .into_iter()
+                .filter(|(index, _)| health[*index].failures == failures)
+                .collect();
+            let fastest = candidates
+                .iter()
+                .filter_map(|(index, _)| health[*index].dial_elapsed)
+                .min();
+            let candidates: Vec<_> = candidates
+                .into_iter()
+                .filter(|(index, _)| {
+                    fastest.is_none_or(|fastest| {
+                        health[*index]
+                            .dial_elapsed
+                            .is_some_and(|elapsed| elapsed <= fastest.mul_f64(1.2))
+                    })
+                })
+                .collect();
+            let mut hash = std::collections::hash_map::DefaultHasher::new();
+            traffic.host.hash(&mut hash);
+            session.hash(&mut hash);
+            return Ok(candidates[hash.finish() as usize % candidates.len()].0);
+        }
+        Err(Error::NoEndpoint)
+    }
+}

+pub(crate) struct Registry {
+    pub routes: Vec<Route>,
+    pub health: RwLock<Vec<Health>>,
+    pub generation: u64,
+    session: u64,
+}

+impl Registry {
+    #[cfg(test)]
+    pub(crate) fn available(routes: Vec<Route>) -> Arc<Self> {
+        Arc::new(Self {
+            health: RwLock::new(
+                routes
+                    .iter()
+                    .map(|_| {
+                        let mut state = Health::new(1);
+                        state.record(&Ok(Duration::from_millis(1)), true, 1);
+                        state
+                    })
+                    .collect(),
+            ),
+            routes,
+            generation: 1,
+            session: 1,
+        })
+    }
+
+    pub async fn prepare(
+        config: &Config,
+        dependencies: &Dependencies,
+        sockets: Arc<Connections>,
+        tasks: TaskTracker,
+        cancel: CancellationToken,
+    ) -> Result<Arc<Self>> {
+        let generation = rand::random();
+        let mut specs = Vec::new();
+        match &config.mode {
+            ModeConfig::Routes(routes) => {
+                for reverse in &routes.reverse {
+                    if reverse.backend.payload() != routes.payload {
+                        continue;
+                    }
+                    let proxy = crate::outbound::ReverseProxy {
+                        backend: reverse.backend,
+                    };
+                    specs.push((
+                        ModeConfig::from(reverse.clone()),
+                        if reverse.backend == RemoteBackend::Relay {
+                            Tier::A
+                        } else {
+                            Tier::B
+                        },
+                        reverse.backend.payload(),
+                        proxy.allows_download(),
+                    ));
+                }
+                if routes.direct {
+                    specs.push((
+                        ModeConfig::Routes(crate::Routes {
+                            payload: routes.payload,
+                            reverse: vec![],
+                            direct: true,
+                            tunnel: vec![],
+                        }),
+                        Tier::C,
+                        routes.payload,
+                        true,
+                    ));
+                }
+                for protocol in &routes.tunnel {
+                    specs.push((
+                        ModeConfig::ForwardProxy {
+                            protocol: protocol.clone(),
+                        },
+                        Tier::D,
+                        routes.payload,
+                        true,
+                    ));
+                }
+            }
+            ModeConfig::LocalReverse {
+                tls_server_name,
+                endpoints,
+            } => {
+                let endpoints = if endpoints.is_empty() {
+                    &config.runtime.discovery_endpoints
+                } else {
+                    endpoints
+                };
+                for endpoint in endpoints {
+                    specs.push((
+                        ModeConfig::LocalReverse {
+                            tls_server_name: tls_server_name.clone(),
+                            endpoints: vec![endpoint.clone()],
+                        },
+                        Tier::A,
+                        Payload::Http,
+                        true,
+                    ));
+                }
+            }
+            _ => specs.push((
+                config.mode.clone(),
+                match &config.mode {
+                    ModeConfig::RemoteReverse {
+                        backend: RemoteBackend::Relay,
+                        ..
+                    } => Tier::A,
+                    ModeConfig::RemoteReverse { .. } => Tier::B,
+                    ModeConfig::ForwardProxy { .. } => Tier::D,
+                    _ => Tier::A,
+                },
+                config.payload(),
+                match &config.mode {
+                    ModeConfig::RemoteReverse { backend, .. } => {
+                        crate::outbound::ReverseProxy { backend: *backend }.allows_download()
+                    }
+                    _ => true,
+                },
+            )),
+        }
+        let mut routes = Vec::new();
+        for (mode, tier, payload, downloads) in specs {
+            let mut child = config.clone();
+            child.mode = mode;
+            let endpoint = match &child.mode {
+                ModeConfig::Routes(_) => None,
+                ModeConfig::ForwardProxy { protocol } => {
+                    let (host, port) = protocol.endpoint();
+                    Some(SocketEndpoint {
+                        host: host.into(),
+                        port,
+                    })
+                }
+                _ => {
+                    let endpoints = child.mode.connection_endpoints();
+                    let first = endpoints
+                        .first()
+                        .or(config.runtime.discovery_endpoints.first())
+                        .ok_or(Error::NoEndpoint)?;
+                    let (host, port) = crate::config::parse_endpoint(first)?;
+                    Some(SocketEndpoint { host, port })
+                }
+            };
+            let observation =
+                crate::ObservationContext::new(child.kind(), dependencies.observer.clone());
+            let mut backend = Backend::create(
+                &child,
+                dependencies.dialer.clone(),
+                sockets.clone(),
+                None,
+                cancel.clone(),
+                observation,
+            )
+            .await?;
+            backend.direct =
+                crate::outbound::DirectOutbound::new(dependencies.dialer.clone(), generation);
+            #[cfg(feature = "reverse-proxy")]
+            {
+                backend.http_cache = dependencies.http_cache.clone();
+            }
+            routes.push(Route {
+                tier,
+                payload,
+                downloads,
+                endpoint,
+                backend: Arc::new(backend),
+            });
+        }
+        let registry = Arc::new(Self {
+            health: RwLock::new(routes.iter().map(|_| Health::new(generation)).collect()),
+            routes,
+            generation,
+            session: rand::random(),
+        });
+        registry
+            .check(dependencies.dialer.clone(), &cancel, true)
+            .await?;
+        if !registry
+            .health
+            .read()
+            .iter()
+            .any(|state| state.eligibility == Eligibility::Available)
+        {
+            return Err(Error::NoEndpoint);
+        }
+        let running = registry.clone();
+        let dialer = dependencies.dialer.clone();
+        tasks.spawn(async move {
+            loop {
+                tokio::select! {
+                    biased;
+                    _ = cancel.cancelled() => break,
+                    _ = tokio::time::sleep(Duration::from_millis(250)) => {}
+                }
+                if running.check(dialer.clone(), &cancel, false).await.is_err()
+                    && cancel.is_cancelled()
+                {
+                    break;
+                }
+            }
+        });
+        Ok(registry)
+    }
+
+    async fn check(
+        &self,
+        dialer: Arc<dyn crate::TcpDialer>,
+        cancel: &CancellationToken,
+        startup: bool,
+    ) -> Result<()> {
+        let deadline = Instant::now() + Duration::from_secs(15);
+        let mut groups: HashMap<SocketEndpoint, Vec<usize>> = HashMap::new();
+        {
+            let mut health = self.health.write();
+            for (index, route) in self.routes.iter().enumerate() {
+                match &route.endpoint {
+                    None => health[index].eligibility = Eligibility::Available,
+                    Some(endpoint) if startup || health[index].due <= Instant::now() => {
+                        groups.entry(endpoint.clone()).or_default().push(index);
+                    }
+                    _ => {}
+                }
+            }
+        }
+        let scheduler = ProbeScheduler {
+            runner: Arc::new(ProbeRunner {
+                probe: TcpProbe { dialer },
+                timeout: Duration::from_secs(3),
+            }),
+            concurrency: 3,
+        };
+        for result in scheduler
+            .run(groups.keys().cloned().collect(), deadline, cancel)
+            .await?
+        {
+            let mut health = self.health.write();
+            for index in &groups[&result.target] {
+                health[*index].record_at(
+                    &result.result,
+                    startup,
+                    self.generation,
+                    result.completed,
+                );
+            }
+        }
+        if cancel.is_cancelled() {
+            Err(Error::Canceled)
+        } else {
+            Ok(())
+        }
+    }
+
+    pub fn select(
+        &self,
+        traffic: &Traffic<'_>,
+        attempt: &mut Attempt,
+    ) -> Result<(usize, Arc<Backend>)> {
+        let index = Planner.select(
+            traffic,
+            &self.routes,
+            &self.health.read(),
+            attempt,
+            self.session,
+        )?;
+        attempt.tried.insert(index);
+        Ok((index, self.routes[index].backend.clone()))
+    }
+
+    pub fn usable(&self, index: usize, traffic: &Traffic<'_>) -> bool {
+        self.health.read()[index].usable(&self.routes[index], traffic, Instant::now())
+    }
+
+    pub fn failure(&self, index: usize, host: &str, error: &Error) {
+        self.failure_at(index, host, error, false);
+    }
+
+    pub fn failure_at(&self, index: usize, host: &str, error: &Error, target: bool) {
+        if matches!(error, Error::Canceled) {
+            return;
+        }
+        let route = &self.routes[index];
+        let target_failure = target
+            || route.tier == Tier::C
+            || route
+                .backend
+                .config
+                .edgeone_origin(host)
+                .is_ok_and(|alias| alias.is_some())
+            || (route.tier == Tier::D
+                && matches!(error, Error::Operation(failure)
+                if failure.stage == crate::Stage::Connect && matches!(failure.code,
+                    crate::FailureCode::Certificate | crate::FailureCode::TlsProtocol)));
+        let mut health = self.health.write();
+        if let Error::Operation(failure) = error {
+            if matches!(failure.stage, crate::Stage::Connect | crate::Stage::Resolve)
+                && !matches!(
+                    failure.code,
+                    crate::FailureCode::Certificate
+                        | crate::FailureCode::TlsProtocol
+                        | crate::FailureCode::HttpStatus
+                )
+                && !target_failure
+                && let Some(endpoint) = &self.routes[index].endpoint
+            {
+                for (route, state) in self.routes.iter().zip(health.iter_mut()) {
+                    if route.endpoint.as_ref() == Some(endpoint) {
+                        state.record(&Err(error.clone()), false, self.generation);
+                    }
+                }
+                return;
+            }
+        }
+        let state = &mut health[index];
+        match error {
+            Error::Credential => state.isolated = true,
+            Error::Operation(failure)
+                if matches!(
+                    failure.code,
+                    crate::FailureCode::Certificate
+                        | crate::FailureCode::TlsProtocol
+                        | crate::FailureCode::ProxyProtocol
+                ) && !target_failure =>
+            {
+                state.isolated = true
+            }
+            Error::Operation(failure) if failure.code == crate::FailureCode::HttpStatus => {}
+            _ => {
+                state
+                    .targets
+                    .insert(host.into(), Instant::now() + Duration::from_secs(15));
+            }
+        }
+    }
+
+    pub fn access_success(&self, index: usize) {
+        let mut health = self.health.write();
+        for (route, state) in self.routes.iter().zip(health.iter_mut()) {
+            if route.endpoint == self.routes[index].endpoint {
+                state.evidence = Instant::now();
+                state.updated = state.evidence;
+                state.failures = 0;
+            }
+        }
+    }

+struct Dialer {
+    calls: Arc<Mutex<Vec<String>>>,
+    fail: bool,
+}
+
+#[async_trait]
+impl TcpDialer for Dialer {
+    async fn connect(&self, host: &str, _: u16) -> Result<BoxStream> {
+        self.calls.lock().push(host.into());
+        if self.fail {
+            return Err(Error::Transport);
+        }
+        let (stream, peer) = tokio::io::duplex(128);
+        drop(peer);
+        Ok(Box::new(stream))
+    }
+
+    async fn resolve(&self, _: &str, port: u16) -> Result<Vec<SocketAddr>> {
+        Ok(vec![SocketAddr::from(([192, 0, 2, 1], port))])
+    }
+}
+
+fn whitelist() -> Whitelist {
+    Whitelist::new(BTreeMap::from([(
+        "app.example.test".into(),
+        "origin.example.test".into(),
+    )]))
+    .unwrap()
+}
+
+fn direct(payload: Payload) -> Routes {
+    Routes {
+        payload,
+        reverse: vec![],
+        direct: true,
+        tunnel: vec![],
+    }
+}

+struct HttpDialer {
+    calls: Arc<Mutex<Vec<String>>>,
+    requests: Arc<Mutex<Vec<String>>>,
+}
+
+#[async_trait]
+impl TcpDialer for HttpDialer {
+    async fn resolve(&self, host: &str, port: u16) -> Result<Vec<SocketAddr>> {
+        assert_eq!(host, "origin.example.test");
+        Ok(vec![SocketAddr::from(([192, 0, 2, 2], port))])
+    }
+
+    async fn connect(&self, host: &str, port: u16) -> Result<BoxStream> {
+        if host == "cdn.example.test" {
+            self.calls.lock().push(host.into());
+            let (stream, peer) = tokio::io::duplex(16);
+            drop(peer);
+            return Ok(Box::new(stream));
+        }
+        assert_eq!((host, port), ("192.0.2.2", 80));
+        self.calls.lock().push(host.into());
+        let (stream, mut server) = tokio::io::duplex(8192);
+        let requests = self.requests.clone();
+        tokio::spawn(async move {
+            let mut bytes = Vec::new();
+            while !bytes.ends_with(b"\r\n\r\n") {
+                match server.read_u8().await {
+                    Ok(byte) => bytes.push(byte),
+                    Err(_) => return,
+                }
+            }
+            requests.lock().push(String::from_utf8(bytes).unwrap());
+            server.write_all(b"HTTP/1.1 429 Too Many Requests\r\nContent-Length: 0\r\nConnection: close\r\n\r\n").await.unwrap();
+        });
+        Ok(Box::new(stream))
+    }
+}

+struct PendingDialer(Arc<std::sync::atomic::AtomicUsize>);
+
+struct PendingGuard(Arc<std::sync::atomic::AtomicUsize>);
+
+impl Drop for PendingGuard {
+    fn drop(&mut self) {
+        self.0.fetch_sub(1, std::sync::atomic::Ordering::SeqCst);
+    }
+}
+
+#[async_trait]
+impl TcpDialer for PendingDialer {
+    async fn connect(&self, _: &str, _: u16) -> Result<BoxStream> {
+        self.0.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
+        let _guard = PendingGuard(self.0.clone());
+        std::future::pending().await
+    }
+}

+#[cfg(feature = "reverse-proxy")]
+#[tokio::test]
+async fn canceled_start_drains_probes_and_never_accepts_business_before_publication() {
+    let active = Arc::new(std::sync::atomic::AtomicUsize::new(0));
+    let routes = Routes {
+        reverse: vec![modes::RemoteReverse {
+            backend: RemoteBackend::Relay,
+            server_name: "access.example.test".into(),
+        }],
+        ..direct(Payload::Http)
+    };
+    let mode = Factory::new(Dependencies::new().with_dialer(PendingDialer(active.clone())))
+        .create(Profile::new("cancel", routes, whitelist()))
+        .unwrap();
+    let cancel = tokio_util::sync::CancellationToken::new();
+    let (result, ()) = tokio::join!(mode.start(cancel.clone()), async {
+        while active.load(std::sync::atomic::Ordering::SeqCst) == 0 {
+            tokio::task::yield_now().await;
+        }
+        assert!(matches!(
+            mode.request(
+                http::Request::get("https://app.example.test/")
+                    .body(body(""))
+                    .unwrap()
+            )
+            .await,
+            Err(Error::Lifecycle)
+        ));
+        cancel.cancel();
+    });
+    assert!(matches!(result, Err(Error::Canceled)));
+    assert_eq!(active.load(std::sync::atomic::Ordering::SeqCst), 0);
+    mode.close().await.unwrap();
+    assert_eq!(mode.status().await.active_sockets, 0);
+}
